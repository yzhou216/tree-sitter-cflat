Forms that are NOT C♭, and which this grammar therefore rejects.

Several of these appear in the repository's `test-inputs/*.cb` files, which are
lexer-only fixtures: they are valid token streams but not valid programs.

================================================================================
No trailing comma in a struct field list (LIST has no trailing comma)
:error
================================================================================

struct Point {
  x: int,
  parent: &Point,
}

--------------------------------------------------------------------------------

================================================================================
No trailing comma in a parameter list
:error
================================================================================

fn f(x: int,) -> int {}

--------------------------------------------------------------------------------

================================================================================
No trailing comma in a let declaration
:error
================================================================================

fn f() -> int {
  let x: int,;
}

--------------------------------------------------------------------------------

================================================================================
let is function-local; it is not a top-level item
:error
================================================================================

let f : (int, bool) -> int;

--------------------------------------------------------------------------------

================================================================================
There is no "new [type; exp]" form
:error
================================================================================

fn f() -> int {
  arr = new [int; 10];
}

--------------------------------------------------------------------------------

================================================================================
let declarations may not follow a statement
:error
================================================================================

fn f() -> int {
  x = 0;
  let y: int;
}

--------------------------------------------------------------------------------

================================================================================
A block may not contain a let declaration
:error
================================================================================

fn f() -> int {
  if x { let y: int; }
}

--------------------------------------------------------------------------------

================================================================================
Assignment is a statement, so it does not nest inside an expression
:error
================================================================================

fn f() -> int {
  foo(x = 1);
}

--------------------------------------------------------------------------------

================================================================================
Assignment does not chain
:error
================================================================================

fn f() -> int {
  x = y = 1;
}

--------------------------------------------------------------------------------

================================================================================
A statement requires its terminating semicolon
:error
================================================================================

fn f() -> int {
  return 0
}

--------------------------------------------------------------------------------

================================================================================
A function definition requires an explicit return type
:error
================================================================================

fn f() {}

--------------------------------------------------------------------------------

================================================================================
An if or while condition may not be parenthesis-free of its block
:error
================================================================================

fn f() -> int {
  if x x = 0;
}

--------------------------------------------------------------------------------

================================================================================
Identifiers may not start with a digit or an underscore
:error
================================================================================

fn f() -> int {
  _x = 0;
}

--------------------------------------------------------------------------------

================================================================================
A program must contain at least one item
:error
================================================================================

// only a comment

--------------------------------------------------------------------------------
