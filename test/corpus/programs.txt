// Point in the plane with a pointer back to its parent.
struct Point {
  x: int,
  y: int,
  parent: &Point
}

fn square(x: int) -> int {
  return x * x;
}

extern print_num : (int) -> int;

fn main() -> int {
  let x: int, y: int;
  x = square(5);
  y = x * 2 + 1;
  if x > y {
    y = x;
  } else {
    y = x - 1;
  }
  while y > 0 {
    y = y - 1;
  }
  return y;
}

// Arithmetic, relational, logical, and pointer/array operators.
fn main() -> int {
  let p: &Point, i: int, arr: &[int];
  p = new Point;
  i = p.x;             // field access
  i = p.*;             // pointer access
  i = arr[0];          // array access
  arr = [int; 10];
  p = nil;
  i = -i + 3 * 4 / 2;
  i = i == 0 or i != 1 and i < 2;
  if not i >= 2 {
    i = i <= 3 ? i : 4;
  }
  return i;
}
